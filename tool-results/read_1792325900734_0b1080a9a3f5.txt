   583→                    dir="ltr"
   584→                    value={form.licenseNumber}

   589→                  />
   590→                </Field>

   597→                      setForm((f) => ({ ...f, licenseSerial: e.target.value }))

   599→                    className="text-right font-mono"
   600→                  />

   611→                  />

   615→          )}
   616→
   617→          {form.category === "property" && (
   618→            <div className="space-y-3 p-3 rounded-card brand-bg-soft">
   619→              <p className="text-xs font-bold text-brand-text">اطلاعات ملک</p>
   620→              <div className="grid grid-cols-2 gap-2">

   623→                    type="text"
   624→                    dir="ltr"
   625→                    value={form.contractNumber}
   626→                    onChange={(e) =>
   627→                      setForm((f) => ({ ...f, contractNumber: e.target.value }))
   628→                    }
   629→                    className="text-right font-mono"
   630→                  />
   631→                </Field>
   632→                <Field label="کد رهگیری">

   634→                    type="text"
   635→                    dir="ltr"
   636→                    value={form.trackingCode}
   637→                    onChange={(e) =>
   638→                      setForm((f) => ({ ...f, trackingCode: e.target.value }))
   639→                    }
   640→                    className="text-right font-mono"
   641→                  />
   642→                </Field>
   643→              </div>
   644→            </div>
   645→          )}
   646→
   647→          {form.category === "insurance" && (

   748→                          a.download = `${viewDoc.title}-page${imageIdx + 1}.jpg`;

   788→                label="نوع سند"

   792→            )}
   793→            {viewDoc.issueDateJalali && (

   795→                label="تاریخ صدور"
   796→                value={viewDoc.issueDateJalali}

   811→                dir="rtl"
   812→              />
   813→            )}
   814→
   815→            {/* Vehicle-specific in viewer */}
   816→            {viewDoc.category === "vehicle" && (

   818→                {viewDoc.licenseNumber && (

   820→                )}
   821→                {viewDoc.licenseSerial && (
   822→                  <CopyableField label="سریال گواهی" value={viewDoc.licenseSerial} dir="ltr" mono />

   825→                  <CopyableField label="شماره پلاک" value={viewDoc.plateNumber} dir="ltr" mono />

   830→                {viewDoc.engineNumber && (
